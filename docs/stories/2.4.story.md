### Component Specifications

**Field Extraction Pattern:**
```elm
-- Extract all available fields from both spreadsheets
extractAvailableFields : Maybe ProcessedData -> Maybe FileData -> Maybe FileData -> List String
extractAvailableFields previewData masterFile dataFile =
    case previewData of
        Just preview ->
            -- Use processed data if available (preferred)
            extractFieldsFromProcessedData preview
        
        Nothing ->
            -- Fallback to raw FileData headers
            case (masterFile, dataFile) of
                (Just master, Just data) ->
                    List.concat [master.headers, data.headers]
                        |> List.Extra.unique
                _ ->
                    []

-- Default selection: all data spreadsheet fields
defaultFieldSelection : Maybe FileData -> Set String
defaultFieldSelection dataFile =
    case dataFile of
        Just data ->
            Set.fromList data.headers
        Nothing ->
            Set.empty
```

### Step Integration Pattern
[Source: architecture/core-workflows.md - Data Extractor Workflow]

**Step 4 Integration in Wizard:**
```elm
-- SelectFields step in Data Extractor workflow
type SelectFieldsMsg
    = ToggleField String
    | SelectAllFields  
    | ClearAllFields
    | ValidateFieldSelection

-- Field selection integration with existing wizard
updateSelectFields : SelectFieldsMsg -> ExtractorModel -> (ExtractorModel, Cmd Msg)
updateSelectFields msg model =
    case msg of
        ToggleField fieldName ->
            let
                updatedFields = 
                    if Set.member fieldName model.selectedFields then
                        Set.remove fieldName model.selectedFields
                    else
                        Set.insert fieldName model.selectedFields
            in
            ( { model | selectedFields = updatedFields }, Cmd.none )

        SelectAllFields ->
            ( { model | selectedFields = Set.fromList model.availableFields }, Cmd.none )

        ClearAllFields ->
            ( { model | selectedFields = Set.empty }, Cmd.none )

        ValidateFieldSelection ->
            -- Validation happens automatically - at least one field required
            ( model, Cmd.none )
```

**Navigation Integration:**
```elm
-- Step validation for wizard navigation
canNavigateFromSelectFields : ExtractorModel -> Bool
canNavigateFromSelectFields model =
    not (Set.isEmpty model.selectedFields)  -- At least one field required

-- Step initialization when entering from preview
initializeSelectFields : ExtractorModel -> ExtractorModel  
initializeSelectFields model =
    let
        available = extractAvailableFields model.previewData model.masterFile model.dataFile
        defaultSelected = 
            case model.dataFile of
                Just dataFile -> Set.fromList dataFile.headers
                Nothing -> Set.empty
    in
    { model 
        | availableFields = available
        , selectedFields = 
            if Set.isEmpty model.selectedFields then 
                defaultSelected 
            else 
                model.selectedFields  -- Preserve existing selection
    }
```

### Field Source Identification
[Source: architecture/data-models.md - FileData]

**Field Categorization Logic:**
```elm
-- Identify field sources for user clarity
type FieldSource = Master | Data | Both

categorizeField : FileData -> FileData -> String -> FieldSource  
categorizeField masterFile dataFile fieldName =
    let
        inMaster = List.member fieldName masterFile.headers
        inData = List.member fieldName dataFile.headers
    in
    case (inMaster, inData) of
        (True, True) -> Both
        (True, False) -> Master
        (False, True) -> Data
        (False, False) -> Master  -- Fallback, shouldn't happen

-- Display field with source indicator
viewFieldWithSource : FileData -> FileData -> String -> Html SelectFieldsMsg
viewFieldWithSource masterFile dataFile fieldName =
    let
        source = categorizeField masterFile dataFile fieldName
        sourceText = case source of
            Master -> " (Master)"
            Data -> " (Data)" 
            Both -> " (Both)"
    in
    label [ class "field-option__label" ]
        [ text (fieldName ++ sourceText) ]
```
